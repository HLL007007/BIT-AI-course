import matplotlib.pyplot as plt
import numpy as np

# 设置中文字体，防止乱码 (根据系统环境可能需要调整，这里尝试使用常见的黑体)
# Windows系统通常用 'SimHei'，Mac系统可能用 'Arial Unicode MS'
plt.rcParams['font.sans-serif'] = ['SimHei', 'Arial Unicode MS', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False  # 解决负号显示问题

# 1. 准备数据
freq_khz = np.array([10, 20, 30, 40, 50])  # 频率 (kHz)

# 阻抗数据 (单位: kΩ) - 直接从表格最后一行读取
z_r = np.array([2.00, 2.00, 1.96, 1.96, 1.96])
z_l = np.array([0.0169, 0.0321, 0.0467, 0.0606, 0.0766])
z_c = np.array([0.0155, 0.00794, 0.00539, 0.00410, 0.00332])

# 2. 创建绘图
fig, ax1 = plt.subplots(figsize=(10, 6))

# 3. 绘制电阻 R (使用左侧Y轴)
color_r = 'tab:red'
ax1.set_xlabel('频率 f (kHz)', fontsize=12)
ax1.set_ylabel('电阻 R 阻抗 (kΩ)', color=color_r, fontsize=12)
ax1.plot(freq_khz, z_r, color=color_r, marker='o', label='电阻 R')
ax1.tick_params(axis='y', labelcolor=color_r)
ax1.grid(True, linestyle='--', alpha=0.6)
ax1.set_ylim(0, 2.5)  # 设置左侧Y轴范围，使R的变化更明显

# 4. 绘制电感 L 和电容 C (使用右侧Y轴，因为数值差异巨大)
ax2 = ax1.twinx()
color_l = 'tab:blue'
color_c = 'tab:green'

ax2.set_ylabel('L 和 C 阻抗 (kΩ)', fontsize=12)
ax2.plot(freq_khz, z_l, color=color_l, marker='s', linestyle='--', label='电感 L')
ax2.plot(freq_khz, z_c, color=color_c, marker='^', linestyle='--', label='电容 C')
ax2.tick_params(axis='y')
ax2.set_ylim(0, 0.08) # 设置右侧Y轴范围

# 5. 添加图例和标题
# 合并两个轴的图例
lines_1, labels_1 = ax1.get_legend_handles_labels()
lines_2, labels_2 = ax2.get_legend_handles_labels()
ax1.legend(lines_1 + lines_2, labels_1 + labels_2, loc='upper right')

plt.title('R、L、C 单个元件阻抗频率特性曲线', fontsize=14)

# 6. 显示图形
plt.tight_layout()
plt.show()
