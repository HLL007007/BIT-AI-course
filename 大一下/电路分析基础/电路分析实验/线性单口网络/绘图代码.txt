import matplotlib.pyplot as plt
import numpy as np

# ==========================================
# 1. 设置中文字体，防止图表显示方块乱码
# ==========================================
plt.rcParams['font.sans-serif'] = ['SimHei']  # Windows 用户使用黑体
# plt.rcParams['font.sans-serif'] = ['Arial Unicode MS'] # Mac 用户请取消注释此行，并注释上一行
plt.rcParams['axes.unicode_minus'] = False    # 正常显示负号

# ==========================================
# 2. 录入三张表格的实验数据
# 外特性曲线通常以电流 I 为横坐标(X轴)，电压 V 为纵坐标(Y轴)
# ==========================================

# 表 2.1: 含源线性单口网络
I1_mA = np.array([1.700, 1.165, 0.887, 0.715, 0.600])
V1_V = np.array([1.70, 2.33, 2.66, 2.86, 3.00])

# 表 2.2: 戴维南等效电路
I2_mA = np.array([1.710, 1.175, 0.893, 0.723, 0.606])
V2_V = np.array([1.71, 2.35, 2.68, 2.89, 3.03])

# 表 2.3: 诺顿等效电路
I3_mA = np.array([1.700, 1.170, 0.890, 0.720, 0.604])
V3_V = np.array([1.70, 2.34, 2.67, 2.88, 3.02])


# ==========================================
# 3. 绘制图表
# ==========================================
plt.figure(figsize=(10, 7)) # 设置画布大小

# 绘制表2.1：蓝色、实线、圆形标记
plt.plot(I1_mA, V1_V, marker='o', linestyle='-', color='blue', 
         linewidth=2, markersize=8, alpha=0.7, label='含源线性单口网络 (表2.1)')

# 绘制表2.2：红色、虚线、方形标记 (稍微错开一点透明度防止完全遮挡)
plt.plot(I2_mA, V2_V, marker='s', linestyle='--', color='red', 
         linewidth=2, markersize=7, label='戴维南等效电路 (表2.2)')

# 绘制表2.3：绿色、点划线、三角形标记
plt.plot(I3_mA, V3_V, marker='^', linestyle='-.', color='green', 
         linewidth=2, markersize=8, label='诺顿等效电路 (表2.3)')

# ==========================================
# 4. 图表装饰与美化
# ==========================================
plt.title('图 2.7：单口网络及其等效电路的外特性曲线比较', fontsize=16)
plt.xlabel('电流 $I_{AB}$ (mA)', fontsize=14)
plt.ylabel('电压 $V_{AB}$ (V)', fontsize=14)

# 添加网格线，方便观察数据点位置
plt.grid(True, linestyle=':', alpha=0.8, color='gray')

# 添加图例，放置在右上角
plt.legend(loc='upper right', fontsize=12)

# 设置坐标轴范围，稍微留出一些边距让图表更好看
plt.xlim(0.4, 2.0)
plt.ylim(1.5, 3.2)

# 反转 X 轴 (可选)：因为实验中电阻 RL 从大到小，电流是从小到大。
# 默认从小到大画没问题，这里保持默认。

# 显示绘制结果
plt.show()
